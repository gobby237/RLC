\subsection{Dati cursore}

Si stima il periodo delle oscillazioni dividendo per 4 la differenza tra il tempo relativo all'ultimo zero decrescente e al primo, propagando le incertezze di lettura dei tempi: 

$$T=t^{-}_{5.5}- t^+_{1.5}=(7.60\pm 0.01) \mu s $$

Si può di conseguenza stimare la pulsazione delle oscillazioni come segue: 

$$\Omega=\frac{2\pi}{T}= (8.27\pm0.02)\cdot10^{5}rad/s$$

Al fine di migliorare l'accuratezza delle misure di tensione, è stato determinato un offset strumentale pari a $0.02 \, \text{V}$ analizzando la regione di baseline del segnale precedente all'innesco della scarica. Tale valore, agente come una componente sistematica sui dati è totalmente compatibile con l'errore di scala sulle tensioni (0.5V f.s.), si è deciso di correggere i dati grezzi riportati in tabella per la linearizzazione successiva.

I valori relativi ai fit effettuati separatamente per i massimi  e i minimi della curva di smorzamento risultano compatibili entro incertezza sperimentale: $\lambda_m=0.13$ per le pendenze e $\lambda_{V_0}=0.11$ per le intercette. Essendo entrambi i $\lambda < 1$ è assicurata un'ottima compatibilità tra i due set di dati, si procede quindi considerando i risultati della regressione combinando tutti i valori in modulo. \\
Dalla pendenza $m =(\delta T)$ si può ricavare $\tau = \frac{1}{\delta}$ usando il periodo $T$ già determinato. \\ Si procede anche con la stima della massima ampiezza di oscillazione $V_1$ come il valore assunto dalla tensione con ascissa $n = 1$. L'incertezza associata è stata ricavata dalla formula di propagazione degli errori tenendo conto della convarianza tra i parametri della regressione. 

$$\tau_{lin}=\frac{T}{m}=(9.2\, \pm \,  0.2) \mathrm{\mu s}$$
$$ V_1 = (2.38 \, \pm \, 0.03) \mathrm{V} $$



Si calcolano a questo punto i rimanenti parametri del circuito, ottenuti tramite le formule riportate in appendice (dalla (16) alla (19)) e i cui risultati sono mostrati nelle tabelle (3) e (4) nelle conclusioni. Le incertezze per ciascuna delle stime si sono ottenute propagando gli errori casuali.

\section{Appendice}

\subsection{Formule statistiche}
\vspace{0.2cm}

\begin{itemize}

\item \textbf{Media aritmetica}
\begin{equation}
    \bar{x} = \frac{1}{N} \sum_{i=1}^{N} x_i
\end{equation}

\item \textbf{Deviazione standard della singola misura} 
\begin{equation}
    \sigma = \sqrt{\frac{1}{N-1} \sum_{i=1}^{N} (x_i - \bar{x})^2}
\end{equation} 
    
\item \textbf{Deviazione standard della media per misure statisticamente indipendenti}
\begin{equation}
    \sigma_{\bar{x}} = \frac{\sigma}{\sqrt{N}}
\end{equation}

\item \textbf{Compatibilità tra due stime}
\begin{equation}
    \lambda = \frac{|a - b|}{\sqrt{\sigma_a^2 + \sigma_b^2}}  
\end{equation}


    \item $\lambda < 1$: compatibilità ottima
    \item $1 \leq \lambda < 2$: compatibilità buona
    \item $2 \leq \lambda < 3$: compatibilità sufficiente
    \item $\lambda \geq 3$: incompatibilità

\item \textbf{Interpolazione lineare \( y = a + bx \) con:}
\begin{itemize}
    \item \( x_i \) statisticamente indipendenti tra loro, \( y_i \) statisticamente indipendenti tra loro;
    \item \( \sigma_{x} \ll \sigma_{y} \);
    \item \( \sigma_i \neq \sigma_y \).
\end{itemize}

\begin{equation}
a = \frac{1}{\Delta} \left[ \left( \sum_{i} \frac{x_i^2}{\sigma_i^2} \right) \cdot \left( \sum_{i} \frac{y_i}{\sigma_i^2} \right) - \left( \sum_{i} \frac{x_i}{\sigma_i^2} \right) \cdot \left( \sum_{i} \frac{x_i y_i}{\sigma_i^2} \right) \right]
\end{equation}

\begin{equation}
b = \frac{1}{\Delta} \left[ \left( \sum_{i} \frac{1}{\sigma_i^2} \right) \cdot \left( \sum_{i} \frac{x_i y_i}{\sigma_i^2} \right) - \left( \sum_{i} \frac{x_i}{\sigma_i^2} \right) \cdot \left( \sum_{i} \frac{y_i}{\sigma_i^2} \right) \right]
\end{equation}

\begin{equation}
\begin{cases}
\sigma_a^2 = \dfrac{1}{\Delta} \sum_{i} \dfrac{x_i^2}{\sigma_i^2}, \\
\sigma_b^2 = \dfrac{1}{\Delta} \sum_{i} \dfrac{1}{\sigma_i^2}.
\end{cases}
\end{equation}

\begin{equation}
\Delta = \left( \sum_{i} \frac{1}{\sigma_i^2} \right) \cdot \left( \sum_{i} \frac{x_i^2}{\sigma_i^2} \right) - \left( \sum_{i} \frac{x_i}{\sigma_i^2} \right)^2
\end{equation}

\begin{equation}
\sigma_{y,\text{post}} = \sqrt{\frac{\sum_{i=1}^N (y_i - a - bx_i)^2}{N - 2}}
\end{equation}

\vspace{0.2cm}

\item \textbf{Somma in quadratura (contributi indipendenti)}
\begin{equation}
\sigma_{\text{tot}} = \sqrt{\sum_{k=1}^{m} \sigma_k^2}
\end{equation}

\item \textbf{Propagazione degli errori (forma generale)}
Dato \(f=f(x_1,\dots,x_m)\), per incertezze piccole:
\begin{equation}
\sigma_f^2 \simeq \sum_{i=1}^{m}\left(\frac{\partial f}{\partial x_i}\right)^2\sigma_{x_i}^2
\;+\;2\sum_{i<j}\left(\frac{\partial f}{\partial x_i}\right)\left(\frac{\partial f}{\partial x_j}\right)\mathrm{cov}(x_i,x_j).
\end{equation}

\item \textbf{Media pesata e incertezza (pesi \(w_i=1/\sigma_i^2\))}
\begin{equation}
\bar{x}_w=\frac{\sum_{i=1}^{N} w_i x_i}{\sum_{i=1}^{N} w_i},
\qquad
\sigma_{\bar{x}_w}=\left(\sum_{i=1}^{N} w_i\right)^{-1/2}.
\end{equation}

\item \textbf{Test di consistenza di \(N\) stime (chi-quadro su medie pesate)}
Per stime \(x_i\) con incertezze \(\sigma_i\) e media pesata \(\bar{x}_w\):
\begin{equation}
\chi^2_{\text{tot}}=\sum_{i=1}^{N}\left(\frac{x_i-\bar{x}_w}{\sigma_i}\right)^2,
\qquad
\nu=N-1.
\end{equation}

\item \textbf{Massima verosimiglianza (MLE) con errori gaussiani indipendenti}
Se \(y_i \sim \mathcal{N}(f(x_i;\boldsymbol{\theta}),\,\sigma_i^2)\) indipendenti:
\begin{equation}
\mathcal{L}(\boldsymbol{\theta})=\prod_{i=1}^{N}\frac{1}{\sqrt{2\pi}\sigma_i}\exp\!\left[-\frac{(y_i-f(x_i;\boldsymbol{\theta}))^2}{2\sigma_i^2}\right].
\end{equation}
Massimizzare \(\mathcal{L}\) equivale a minimizzare la \(\chi^2\):
\begin{equation}
-2\ln\mathcal{L}(\boldsymbol{\theta})=\chi^2(\boldsymbol{\theta})+\text{costante}.
\end{equation}

\item \textbf{Matrice di covarianza dei parametri (approssimazione lineare)}
Indicando con \(J\) la jacobiana \(J_{i\mu}=\partial f(x_i;\boldsymbol{\theta})/\partial\theta_\mu\) e \(W=\mathrm{diag}(1/\sigma_i^2)\):
\begin{equation}
\mathrm{cov}(\boldsymbol{\theta}) \simeq (J^{T}WJ)^{-1}.
\end{equation}

\subsection{Formule fisiche}

\item \textbf{Pulsazione di risonanza}
\begin{equation}
    \omega_0=\sqrt{\omega_d^2+\frac{1}{\tau^2}}
\end{equation}

\item \textbf{Fattore di merito}
\begin{equation}
    Q=\frac{\omega_0\tau}{2}
\end{equation}

\item \textbf{Induttanza del circuito}
\begin{equation}
    L=\frac{1}{\omega_0^2C}
\end{equation}

\item \textbf{Resistenza totale del circuito}
\begin{equation}
    R_{tot}=\frac{2L}{\tau}
\end{equation}


\end{itemize}
